Struct

  • Immutable fields.
  • Methods return new structs rather than mutating in place.
  • Declare with let. Declaring with var is an error.
struct Point() do
    let float x = 0
    let float y = 0
    Point move(dx: float, dy: float) do
        return Point(this.x + dx, this.y + dy)
    end
end

let Point p1 = Point(0.0, 0.0)
let Point moved = p1.move(2, 3)

Frame

  • Mutable fields and methods that mutate in place.
  • Can be declared with var or let, but mutating methods require a mutable binding.
frame MovablePoint() do
    var float x = 0
    var float y = 0
    void move(float dx, float dy) do
        this.x = this.x + dx
        this.y = this.y + dy
    end
end

var MovablePoint p2 = MovablePoint(3.0, 4.0)
p2.move(1.0, 1.0)
