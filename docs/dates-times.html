Available functions

  • now() -> date, utcNow() -> date
  • date(y: int, m: int, d: int) -> date
  • dateTime(y: int, m: int, d: int, hh: int, mm: int, ss: int) -> date
  • parseDate(iso: string) -> date (ISO-8601)
  • formatDate(dt: date, fmt: string) -> string (supports %Y %m %d %H %M %S %F %T %z %Z)
  • toUTC(dt: date) -> date, toLocal(dt: date) -> date
  • toTZ(dt: date, hours: int, minutes: int = 0) -> date
  • year(dt) month(dt) day(dt) hour(dt) minute(dt) second(dt) -> int
  • dayOfWeek(dt) -> int
  • daysBetween(a: date, b: date) -> int
  • addDays(dt: date, n: int) -> date, likewise addHours/addMinutes/addSeconds

Current time

let auto today = now()
let auto utc = utcNow()
print("Today: ${today|%F}")
print("UTC: ${utc|%F %T%Z}")

Construction and formatting

let auto bday = date(1990, 7, 6)
let auto meetup = dateTime(2025, 5, 20, 18, 30, 0)
print(formatDate(bday, "%F"))
print(formatDate(meetup, "%F %T"))

Time zones

let auto meet_brt = toTZ(meetup, -3)
print(formatDate(meet_brt, "%F %T%z"))

Field extraction

year(dt) month(dt) day(dt) hour(dt) minute(dt) second(dt)

Arithmetic and comparison

let auto days = daysBetween(meetup, bday)
print(days > 0)
let auto inTwo = addDays(bday, 2)
let auto plusHours = addHours(meetup, 5)
let auto plusMinutes = addMinutes(meetup, 90)
let auto plusSeconds = addSeconds(meetup, 3661)

Parsing

let auto parsed = parseDate("2025-12-31T23:59:59")
print(formatDate(parsed, "%F %T"))

Pattern matching

match parsed do
    dateTime(2025, 12, 31, 23, 59, 59) -> print("NewYearEveEve")
    _ -> print("other")
end
