Available methods

  • On regex
    • test(s: string) -> bool
    • matchFirst(s: string) -> Match
    • matchAll(s: string) -> list<Match>
    • replace(s: string, repl: string) -> string (same as replaceAll)
  • On string
    • matches(r: regex) -> bool
    • matchFirst(r: regex) -> Match
    • matchAll(r: regex) -> list<Match>
    • replace(r: regex, repl: string) -> string
  • Match object properties
    • .hit (string) · .empty (bool)

Construction

Use regex("..."). If the pattern is a string literal, it is validated at compile time.

let auto reg = regex("(\\w+)")

Testing and matching

if reg.test("hello") do
    print("Match found!")
end

let auto match_first = reg.matchFirst("hello world")
print(match_first.hit)
print(match_first.empty)

let auto matches = reg.matchAll("Hello, duende 123!")
print(matches.length)
let auto first_res = matches[0]
print(first_res.hit)

Replacement

let string replaced = reg.replace("hello world", "WORD")
print(replaced) // "WORD world"
