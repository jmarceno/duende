This example uses the bundled std.json provider.

Available functions and types

  • parseJSON(text: string) -> JsonValue
  • JsonValue.str -> string — returns the string value
  • JsonValue.floating -> float — returns the numeric value
  • JsonValue.array -> list<JsonValue> — returns an array of JsonValue
  • JsonValue["key"] -> JsonValue — object key access
  • JsonValue[index] -> JsonValue — array index access
import std.json

let string text = "{\"language\":\"D\",\"list\":[\"a\",\"b\",\"c\",\"D\"],\"rating\":3.5}"

var auto j = parseJSON(text)
print(j["language"].str)
print(j["list"].array.length)
print(j["list"][0].str)
print(j["list"][3].str)
print(j["rating"].floating)
