If / elif / else

if n <= 1 do
    return 1
elif n > 10 do
    return 42
else
    return n * 2
end

While

var int count = 1
while count < 5 do
    print(count)
    count = count + 1
end

Break and Continue

Use continue to skip to the next iteration and break to exit the loop.

var int i = 0
while true do
  i = i + 1
  if i == 3 do
    continue
  end
  print(i)
  if i == 5 do
    break
  end
end

For-in ranges

Use the range operator a .. b. Optional step is not shown in examples.

for i in 1 .. 5 do
    print(i)
end

Boolean operators

!, &&, || are supported (no word forms).

let bool a = true
let bool b = false
print(!(a && b) && (a || b))
