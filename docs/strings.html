Interpolation and formatting

let string name = "Duende"
let int score = 95
let float ratio = 1.0 / 3.0
print("Hello ${name}! Score=${score}, ratio=${ratio|%.2f}")

Manipulation

let list<string> parts = "a,b,c".split(",")
print(parts.join("-"))
print("Hello World".replace("World", "Duende"))
print("hello".toUpperCase())
print("HELLO".toLowerCase())
print("  trim me  ".trim())
print("abcdef".substring(2, 5))

Searching

print("foobar".indexOf("o"))
print("foobar".lastIndexOf("o"))
print("foobar".startsWith("foo"))
print("foobar".endsWith("bar"))
print("foobar".contains("oba"))
print("foo123".matches(regex("\\d+")))

Conversion and encoding

print("42".toInt())
print("3.14".toFloat())
print("true".toBool())
let bytes b = "hi".toBytes()
print(fromBytes(b))

Comparison and formatting helpers

print("abc".compare("abd"))   // -1
print("abc".equals("abc"))    // true
print("abc".notEquals("abd")) // true
print("Value: %04d".format(7))
print("%.1f".formatNumber(2.56))

Utilities

print("".isEmpty())
print("xyz".length())
print("xyz".charAt(1))
print("xyz".charCodeAt(1))

Escaping in interpolation

Use backslash to escape characters inside strings. In addition to \n, \t, \r, \" and \\:

  • \${ inserts a literal ${ without starting interpolation.
  • \} inserts a literal closing brace } without ending interpolation.

Both double-quoted and single-quoted strings support these escapes.

let string name = "Duende"
print("Literal: \\${name} and \\}")     // Literal: ${name} and }
print("Mix: \\${name} -> ${name}")      // Mix: ${name} -> Duende
print('Singles: \\ and \\} and \\${ ok') // Singles: \ and } and ${ ok
