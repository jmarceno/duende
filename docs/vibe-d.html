The vibe_d package provides comprehensive web development capabilities through vibe.d integration. This includes HTTP/HTTPS servers and clients, WebSocket support, SSL/TLS contexts, async operations, logging, file system operations, process management, and timers.

Import

import vibe_d

HTTP Server

Create and manage HTTP/HTTPS servers with automatic routing and request handling.

startServer

string startServer(int port, bool https = false)

Starts an HTTP or HTTPS server on the specified port. Returns a server ID for management.

// Start HTTP server on port 8080
let string serverId = startServer(8080, false)

// Start HTTPS server on port 8443
let string httpsServerId = startServer(8443, true)

The server automatically includes default routes:

  • / — Returns "Hello from Duende vibe.d server!"
  • /api/test — Returns JSON with message and timestamp
  • /api/echo — Echoes POST request body as JSON
  • * — Serves static files from current directory

HTTP Client

Create HTTP clients and perform various HTTP requests.

createHttpClient

string createHttpClient(string baseUrl = "")

Creates an HTTP client, optionally with a base URL. Returns a client ID.

// Create generic client
let string clientId = createHttpClient()

// Create client with base URL
let string apiClient = createHttpClient("https://api.example.com")

HTTP Request Methods

string httpGet(string clientId, string url)
string httpPost(string clientId, string url, string data)
string httpPut(string clientId, string url, string data)
string httpDelete(string clientId, string url)

Perform HTTP requests. POST and PUT methods accept JSON data.

let string clientId = createHttpClient()

// GET request
let string response = httpGet(clientId, "http://example.com/api/users")

// POST request with JSON data
let string postData = "{\"name\": \"John\", \"age\": 30}"
let string postResponse = httpPost(clientId, "http://api.example.com/users", postData)

// PUT request
let string putResponse = httpPut(clientId, "http://api.example.com/users/123", postData)

// DELETE request
let string deleteResponse = httpDelete(clientId, "http://api.example.com/users/123")

HTTPS Request Methods

string httpsGet(string clientId, string url)
string httpsPost(string clientId, string url, string data)

Perform HTTPS requests with automatic TLS context creation.

let string clientId = createHttpClient()

// HTTPS GET request
let string response = httpsGet(clientId, "https://api.github.com/users")

// HTTPS POST request
let string secureData = "{\"message\": \"secure data\"}"
let string httpsResponse = httpsPost(clientId, "https://api.example.com/secure", secureData)

WebSocket

Create WebSocket servers and clients for real-time communication.

createWebSocketServer

string createWebSocketServer(int port, string path = "/ws")

Creates a WebSocket server on the specified port and path. Automatically handles ping/pong and echo functionality.

// Create WebSocket server on port 9001
let string wsServerId = createWebSocketServer(9001, "/websocket")

// Create on default path /ws
let string defaultWsId = createWebSocketServer(8080)

connectWebSocket

string connectWebSocket(string url)

Connects to a WebSocket server. Returns a connection ID.

let string wsConnectionId = connectWebSocket("ws://localhost:9001/websocket")

SSL/TLS

Manage SSL/TLS contexts for secure communications.

createTlsContext

string createTlsContext(int kind = 0)

Creates a TLS context. Kind: 0 for client, 1 for server.

// Create client TLS context
let string clientTls = createTlsContext(0)

// Create server TLS context
let string serverTls = createTlsContext(1)

Async Operations

Handle timers, timeouts, and asynchronous operations.

Timers

string createTimer(int intervalMs, bool repeat = false)
string setTimeout(int timeoutMs)

Create timers and timeouts. Timers can be repeating or one-shot.

// Create repeating timer (every 1 second)
let string timerId = createTimer(1000, true)

// Create one-shot timer
let string oneShot = createTimer(5000, false)

// Set timeout (one-shot, no repeat)
let string timeoutId = setTimeout(2000)

Event Loop

runEventLoop()
exitEventLoop()

Control the event loop for async operations.

// Start event loop (runs for demo period)
runEventLoop()

// Stop event loop
exitEventLoop()

File System

Async file and directory operations.

File Operations

string readFileAsync(string filename)
bool writeFileAsync(string filename, string content)

Asynchronously read and write files.

// Write file
let bool writeSuccess = writeFileAsync("data.txt", "Hello, World!")

// Read file
let string content = readFileAsync("data.txt")

Directory Operations

bool createDirectoryAsync(string dirname)

Asynchronously create directories.

let bool dirSuccess = createDirectoryAsync("new_directory")

Process Management

Spawn and manage external processes.

spawnProcess

string spawnProcess(string command, string[] args = [])

Spawns an external process with optional arguments. Returns a process ID.

// Spawn simple command
let string processId = spawnProcess("ls")

// Spawn with arguments
let string processWithArgs = spawnProcess("ls", ["-la", "/tmp"])

Logging

Comprehensive logging system with multiple levels.

Logger Creation

string createLogger(string name, int level = 3)

Creates a logger with specified name and level. Levels: 0=trace, 1=debug, 2=info, 3=warn, 4=error, 5=critical.

// Create logger with warning level
let string logger = createLogger("MyApp", 3)

// Create debug logger
let string debugLogger = createLogger("Debug", 1)

Logging Methods

logInfo(string message)
logError(string message)
logWarn(string message)
logDebug(string message)

Log messages at different levels.

let string logger = createLogger("MyApp", 2)

logInfo("Application started")
logWarn("This is a warning")
logError("An error occurred")
logDebug("Debug information")

Complete Example

Here's a comprehensive example demonstrating various vibe.d features:

import vibe_d

// 1. Create logger
let string logger = createLogger("WebApp", 2)
logInfo("Starting web application")

// 2. Setup TLS contexts
let string clientTls = createTlsContext(0)
let string serverTls = createTlsContext(1)

// 3. Start HTTP server
let string serverId = startServer(8080, false)
logInfo("HTTP server started on port 8080")

// 4. Create WebSocket server
let string wsId = createWebSocketServer(9001, "/chat")
logInfo("WebSocket server started on port 9001")

// 5. Setup HTTP client
let string clientId = createHttpClient()

// 6. File operations
let bool writeSuccess = writeFileAsync("app.log", "Application started")
if writeSuccess do
    logInfo("Log file created successfully")
end

// 7. Create timers
let string heartbeat = createTimer(30000, true)  // 30 second heartbeat
let string startup = setTimeout(5000)            // 5 second startup delay

// 8. Spawn monitoring process
let string monitor = spawnProcess("ps", ["aux"])

logInfo("Web application initialization complete")

// 9. Test API endpoint
let string response = httpGet(clientId, "http://localhost:8080/api/test")
logInfo("API test completed")
