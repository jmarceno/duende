Available methods

  • Lists
    • add(x: any)
    • slice(start: int, end: int) -> list<T>
    • length property and length() method
    • empty property
    • first / last properties (return -1 or "-1" for strings when empty)
    • sort() (in-place, lexicographic)
    • join(sep: string) -> string
  • Dictionaries
    • has(key: string) -> bool
    • length property and empty property
    • keys -> list<string>, values -> list<string>
    • items() -> list<{key:string, value:string}>

Lists

var list<int> numbers = [1, 2, 3, 4, 5]
numbers.add(6)
let int first = numbers[0]
print(numbers.length)
print(numbers.empty)
let list<int> slice = numbers.slice(1, 4)
let numbers_first = numbers.first
let numbers_last = numbers.last
// In-place sort (lexicographic)
var list<string> names = ["bob", "alice", "carol"]
names.sort()
print(names.first) // "alice"
print(names.last)  // "carol"

Dictionaries

var dict config = {"host": "localhost", "port": 8080}
if config.has("port") do
    let int port = config["port"]
end

print(config.length)
print(config.empty)
let list<string> keys = config.keys
let list<string> values = config.values
