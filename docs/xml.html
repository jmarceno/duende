XML

XML parsing and generation via the std.xml package (powered by dxml).

Available functions and types

  • Type: XmlNode with fields name, attributes: dict, text, children: list<XmlNode>
  • parse(xml: string) -> XmlNode, parseFile(path: string) -> XmlNode
  • find(node: XmlNode, tag: string) -> list<XmlNode>, findAll(node, tag) -> list<XmlNode>
  • getAttr(node, name: string, default: string = "") -> string, setAttr(node, name, value)
  • getText(node) -> string, setText(node, value: string)
  • fromList(parentTag: string, items: list, itemTag: string = "item") -> XmlNode
  • fromData(dict, children: list? = null) -> XmlNode
  • toXml(node) -> string, toXmlPretty(node) -> string

Import

import std.xml as xml

Parse

let root = xml.parse("<root a=\"1\"><child>hi</child></root>")
print(root.name)        # root
print(xml.getAttr(root, "a"))   # 1
print(xml.find(root, "child").len) # 1

Query

let kids = xml.find(root, "child")
let first = kids[0]
print(xml.getText(first))

Modify

xml.setAttr(root, "lang", "en")
xml.setText(first, "hello")

Generate

print(xml.toXml(root))         # compact
print(xml.toXmlPretty(root))  # pretty

Build from data

let items = ["one", "two"]
let listNode = xml.fromList("list", items, "li")
print(xml.toXmlPretty(listNode))

See also the example in examples/std_xml.du.
