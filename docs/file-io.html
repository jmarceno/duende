Paths and directories

let Result<string> cwdRes = getCwd()
let string cwd = cwdRes ? else ""
let string tmpDir = joinPath(cwd, "io_tmp")

if dirExists(tmpDir) do removeDir(tmpDir) end
makeDir(tmpDir)
print(dirExists(tmpDir))

Text files

writeText(file1, "hello")
match readText(file1) do
    Ok(txt) -> print(txt)
end
appendText(file1, "!")
match readText(file1) do
    Ok(txt2) -> print(txt2)
end

writeLines(file2, ["a", "b", "c"])
let list<string> lines = readLines(file2) ? else []

Bytes

let bytes data = [1, 2, 255]
writeBytes(binf, data)
match readBytes(binf) do
    Ok(b) -> print(b.length)
end

Metadata and listing

match fileSize(file1) do
    Ok(sz) -> print(sz)
end
match getMetadata(file1) do
    Ok(meta) -> print(meta["size"]) // dict metadata
end
match listDir(tmpDir) do
    Ok(entries) -> print(entries.length)
end

Deletion

deleteFile(file1)
removeDir(tmpDir)
