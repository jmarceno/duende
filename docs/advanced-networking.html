Timeout Handling

Duende's networking package provides timeout-aware functions to prevent indefinite blocking on I/O operations. These are essential for robust network applications.

TCP timeouts

import std.networking

int main() do
  var auto listener = tcpListen(8080)
  spawnEchoOnce(listener)  // Background echo server
  
  var auto client = tcpConnect("127.0.0.1", 8080)
  tcpSend(client, "hello", true)
  
  // Timeout on receive - returns empty string if timeout elapses
  var string response = tcpRecvLineTimeout(client, 5000)  // 5 second timeout
  if response == "" do
    print("Server didn't respond within 5 seconds")
  else
    print("Got response: " + response)
  end
  
  tcpClose(client)
  tcpClose(listener)
  return 0
end

UDP timeouts

import std.networking

int main() do
  var auto server = udpOpen(0)
  var int port = udpLocalPort(server)
  spawnUdpEchoOnce(server)  // Background echo
  
  var auto client = udpOpen(0)
  udpSendTo(client, "127.0.0.1", port, "ping")
  
  // UDP recv with configurable timeout
  var string reply = udpRecv(client, 2000)  // 2 second timeout
  if reply == "" do
    print("No UDP response within 2 seconds")
  else
    print("UDP reply: " + reply)
  end
  
  udpClose(client)
  udpClose(server)
  return 0
end

Accept timeouts

import std.networking

int main() do
  var auto listener = tcpListen(8080)
  
  // Non-blocking accept with timeout
  var TcpSocket client
  var bool gotConnection = tcpAcceptTimeout(listener, client, 3000)  // 3 seconds
  
  if gotConnection do
    print("Client connected!")
    tcpSend(client, "Welcome!", true)
    tcpClose(client)
  else
    print("No connection within 3 seconds")
  end
  
  tcpClose(listener)
  return 0
end

Platform Nuances

Duende's networking layer abstracts platform differences, but understanding the underlying implementations helps with debugging and performance tuning.

Cross-platform select implementation

The timeout functionality uses different mechanisms on different platforms:

  • POSIX systems (Linux, macOS, BSD): Uses standard select() system call
  • Windows: Uses Winsock's select() implementation
  • Fallback: Simple sleep loop if platform detection fails

Platform-specific code structure

// In duende_net.d helper module
version(Posix) {
    // POSIX select implementation
    import core.sys.posix.sys.select;
    // ... POSIX-specific code
} else version(Windows) {
    // Windows Winsock select implementation
    import core.sys.windows.winsock2;
    // ... Windows-specific code
} else {
    // Fallback implementation
    import core.thread : Thread;
    import core.time : dur;
    // ... fallback polling
}

Socket behavior differences

  • Address binding: POSIX allows immediate reuse of addresses with SO_REUSEADDR, Windows may require SO_REUSEADDR and has different TIME_WAIT behavior
  • Error codes: POSIX uses errno values, Windows uses WSAGetLastError()
  • Blocking behavior: Both platforms support non-blocking sockets, but error handling differs

Multi-Packet Patterns

Real-world network applications often need to handle multiple packets, streaming data, or request-response cycles.

TCP streaming with line protocols

import std.networking

int main() do
  var auto listener = tcpListen(8080)
  print("Server listening on port 8080")
  
  // Accept multiple connections
  while true do
    var TcpSocket client
    var bool gotClient = tcpAcceptTimeout(listener, client, 1000)
    if not gotClient do
      continue  // Try again
    end
    
    print("Client connected")
    
    // Handle multiple lines from this client
    while true do
      var string line = tcpRecvLineTimeout(client, 5000)
      if line == "" do
        print("Client timeout or disconnect")
        break
      end
      
      if line == "quit" do
        tcpSend(client, "Goodbye!", true)
        break
      end
      
      // Echo with prefix
      tcpSend(client, "Echo: " + line, true)
    end
    
    tcpClose(client)
  end
  
  tcpClose(listener)
  return 0
end

UDP message patterns

import std.networking

int main() do
  var auto server = udpOpen(9090)
  print("UDP server on port 9090")
  
  // Handle multiple UDP messages
  while true do
    var string host
    var int port
    var string msg = udpRecvFrom(server, host, port)
    
    if msg == "" do
      continue  // Keep listening
    end
    
    print("From " + host + ":" + string(port) + " -> " + msg)
    
    if msg == "ping" do
      udpSendTo(server, host, port, "pong")
    elif msg == "time" do
      udpSendTo(server, host, port, "TODO: current time")
    elif msg == "quit" do
      udpSendTo(server, host, port, "bye")
      break
    else
      udpSendTo(server, host, port, "unknown: " + msg)
    end
  end
  
  udpClose(server)
  return 0
end

Request-response with correlation

import std.networking

// Simple request-response pattern
int sendRequest(TcpSocket sock, string request) do
  tcpSend(sock, request, true)
  var string response = tcpRecvLineTimeout(sock, 3000)
  if response == "" do
    print("Request timeout: " + request)
    return -1
  end
  print("Request: " + request + " -> Response: " + response)
  return 0
end

int main() do
  var auto listener = tcpListen(8080)
  spawnEchoOnce(listener)  // Simple echo server
  
  var auto client = tcpConnect("127.0.0.1", 8080)
  
  // Send multiple requests in sequence
  sendRequest(client, "hello")
  sendRequest(client, "world")
  sendRequest(client, "duende")
  
  tcpClose(client)
  tcpClose(listener)
  return 0
end

Error Handling & Debugging

Network operations can fail in various ways. Here are common patterns for robust error handling:

Connection failures

import std.networking

int main() do
  // Try to connect to a service that might not be running
  var auto sock = tcpConnect("127.0.0.1", 12345)  // May fail
  
  // In real code, you'd want to handle connection errors
  // For now, Duende doesn't expose detailed error info
  // But you can implement retry logic:
  
  var int retries = 3
  while retries > 0 do
    // Attempt operation
    tcpSend(sock, "test", true)
    var string response = tcpRecvLineTimeout(sock, 1000)
    
    if response != "" do
      print("Success: " + response)
      break
    end
    
    retries = retries - 1
    print("Retry remaining: " + string(retries))
  end
  
  tcpClose(sock)
  return 0
end

Graceful shutdown

import std.networking

int main() do
  var auto listener = tcpListen(8080)
  var bool shutdown = false
  
  // In a real application, you'd handle signals
  // For demo purposes, we'll use a counter
  var int connections = 0
  
  while not shutdown do
    var TcpSocket client
    var bool gotClient = tcpAcceptTimeout(listener, client, 1000)
    
    if gotClient do
      connections = connections + 1
      print("Connection #" + string(connections))
      
      tcpSend(client, "Hello client #" + string(connections), true)
      tcpClose(client)
      
      // Shutdown after 3 connections for demo
      if connections >= 3 do
        shutdown = true
      end
    end
  end
  
  print("Shutting down gracefully")
  tcpClose(listener)
  return 0
end

Performance Considerations

  • Buffer sizes: D's socket implementation uses reasonable defaults, but large transfers may benefit from chunked reading
  • Connection pooling: For high-frequency client applications, consider reusing connections
  • Timeout tuning: Shorter timeouts improve responsiveness but may cause false failures on slow networks
  • Select limitations: The select-based timeout implementation has platform-specific limits on file descriptor numbers

Best Practices

  • Always use timeouts for network operations in production code
  • Implement proper error handling and retry logic
  • Close sockets explicitly to avoid resource leaks
  • Use background helpers (spawnEchoOnce, etc.) only for testing and demos
  • Test your networking code on target platforms, especially if targeting both POSIX and Windows
  • Consider using UDP for fire-and-forget messaging, TCP for reliable data transfer

For basic networking usage, see the main Networking documentation. The networking package is built-in to Duende and requires no additional dependencies.
